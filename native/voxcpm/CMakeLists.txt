cmake_minimum_required(VERSION 3.14...3.28)
project(reccut_voxcpm C CXX)
if(NOT VOXCPM_SOURCE_DIR)
  message(FATAL_ERROR "Supply the pinned llama.cpp-omni checkout with VOXCPM_SOURCE_DIR")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
add_subdirectory("${VOXCPM_SOURCE_DIR}" upstream)
find_package(Threads REQUIRED)
add_executable(reccut-voxcpm voxcpm_server.cpp)
set_target_properties(reccut-voxcpm PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_compile_features(reccut-voxcpm PRIVATE cxx_std_17)
target_include_directories(reccut-voxcpm PRIVATE "${VOXCPM_SOURCE_DIR}/vendor")
target_link_libraries(reccut-voxcpm PRIVATE voxcpm2_runtime llama-common ggml Threads::Threads)
